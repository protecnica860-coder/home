function doGet(e) {
  const action = e.parameter.action || "getLeads";
  if (action === "getLeads") {
    return ContentService.createTextOutput(JSON.stringify(getLeads()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("Invalid action");
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents || "{}");
  const action = data.action || "createLead";
  if (action === "createLead") {
    return ContentService.createTextOutput(JSON.stringify(createLead(data.payload)))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (action === "updateLead") {
    return ContentService.createTextOutput(JSON.stringify(updateLead(data.payload)))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("Invalid action");
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Leads");
  if (!sheet) {
    sheet = ss.insertSheet("Leads");
    sheet.appendRow([
      "id", "created_at", "nombre", "empresa", "telefono", "email", "ciudad_comuna",
      "tipo_solicitud", "necesita", "canal_origen", "utm_source", "utm_medium",
      "utm_campaign", "utm_term", "utm_content", "vendedor_asignado", "estado",
      "proxima_gestion", "monto_cotizacion", "notas_internas"
    ]);
  }
  return sheet;
}

function getLeads() {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  return data.map(row => {
    const item = {};
    headers.forEach((key, i) => item[key] = row[i]);
    return item;
  });
}

function createLead(payload) {
  const sheet = getSheet();
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const row = headers.map(key => payload[key] || "");
  sheet.appendRow(row);
  return { success: true };
}

function updateLead(payload) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  const idIndex = headers.indexOf("id");
  const rowIndex = data.findIndex(row => row[idIndex] === payload.id);
  if (rowIndex === -1) return { success: false };
  const updateRow = headers.map(key => payload[key] || "");
  sheet.getRange(rowIndex + 2, 1, 1, updateRow.length).setValues([updateRow]);
  return { success: true };
}
